#### StateManager - 状态管理核心
**职责**: 统一状态管理、原子性操作、状态同步、性能优化

```javascript
// 状态管理核心类
class StateManager {
  constructor() {
    this.stateCache = null;
    this.pendingOperations = new Map();
    this.listeners = new Set();
    this.initialized = false;
    
    // 性能优化属性
    this.batchUpdateQueue = [];
    this.batchUpdateTimer = null;
    this.batchUpdateDelay = 50;
    this.storageAccessCount = 0;
    this.cacheHitCount = 0;
    this.lastPerformanceReport = Date.now();
    this.debouncedOperations = new Map();
    
    // 监控管理器
    this.monitoringManager = new MonitoringManager();
  }
  
  // 初始化状态管理器
  async initialize() {
    await this.preloadState();
    this.setupStorageListener();
    this.initialized = true;
  }
  
  // 获取完整插件状态（带缓存）
  async getPluginState() {
    this.cacheHitCount++;
    return this.stateCache || await this.loadStateFromStorage();
  }
  
  // 原子性状态更新（带批处理）
  async updatePluginState(changes, operationId) {
    return await this.performStateUpdate(changes, operationId);
  }
  
  // 批处理状态更新
  queueBatchUpdate(changes) {
    this.batchUpdateQueue.push(changes);
    this.scheduleBatchUpdate();
  }
  
  // 性能监控方法
  getPerformanceMetrics() {
    return {
      storageAccessCount: this.storageAccessCount,
      cacheHitRate: this.cacheHitCount / Math.max(this.storageAccessCount, 1),
      pendingOperations: this.pendingOperations.size,
      batchQueueSize: this.batchUpdateQueue.length,
      debouncedOperations: this.debouncedOperations.size,
      memoryUsage: this.getMemoryUsage()
    };
  }
  
  // 状态变更监听器
  addStateChangeListener(listener) {
    this.listeners.add(listener);
  }
  
  // 状态变更通知
  async notifyStateChange(changes, operationId) {
    await this.monitoringManager.logStateChange(changes, operationId);
    this.listeners.forEach(listener => listener(changes, operationId));
  }
}
```

    this.iconManager = new IconManager();

    this.stateConfirmationManager = new StateConfirmationManager();

    await this.stateManager.initialize();
    const state = await this.stateManager.getPluginState();
    await this.iconManager.updateIcon(state);
  }
  
  // 消息路由处理（集成所有管理器）
  setupMessageHandlers() {
    chrome.runtime.onMessage.addListener(async (message, sender, sendResponse) => {
      try {
        switch (message.type) {
          case 'GET_PERFORMANCE_METRICS':
            return this.stateManager.getPerformanceMetrics();
          case 'GET_MONITORING_DATA':
            return this.stateManager.getMonitoringData();
          case 'PERFORM_HEALTH_CHECK':
            return this.stateManager.performHealthCheck();
          case 'EXPORT_MONITORING_LOGS':
            return this.stateManager.exportMonitoringLogs();
          case 'RESET_MONITORING_METRICS':
            return this.stateManager.resetMonitoringMetrics();
          // ... 其他消息处理
        }
      } catch (error) {
        console.error('Message handling error:', error);
        return { success: false, error: error.message };
      }
    });
  }
  
  // 标签页监听（自动启用管理）
  setupTabsListener() {
    chrome.tabs.onUpdated.addListener(async (tabId, changeInfo, tab) => {
      if (changeInfo.status === 'complete' && tab.url) {
        await this.autoEnableManager.handleTabUpdate(tabId, tab.url);
      }
    });
  }
}

### 3. 状态确认管理器 (background/state-confirmation-manager.js)
**职责**: 状态变更确认、请求追踪、自动恢复

```javascript
class StateConfirmationManager {
  constructor() {
    this.pendingRequests = new Map();
    this.requestTimeout = 5000;
    this.maxRetries = 3;
    this.statistics = {
      totalRequests: 0,
      successfulRequests: 0,
      failedRequests: 0,
      timeoutRequests: 0,
      retriedRequests: 0
    };
  }
  
  // 创建确认请求
  async createConfirmationRequest(operation, data, options = {}) {
    const requestId = this.generateRequestId();
    const request = {
      id: requestId,
      operation,
      data,
      timestamp: Date.now(),
      retryCount: 0,
      maxRetries: options.maxRetries || this.maxRetries,
      timeout: options.timeout || this.requestTimeout,
      status: 'pending'
    };
    
    this.pendingRequests.set(requestId, request);
    this.statistics.totalRequests++;
    
    return requestId;
  }
  
  // 确认请求完成
  confirmRequest(requestId, result) {
    const request = this.pendingRequests.get(requestId);
    if (request) {
      request.status = 'completed';
      request.result = result;
      this.statistics.successfulRequests++;
      this.pendingRequests.delete(requestId);
    }
  }
  
  // 状态验证和恢复
  async validateAndRecover(expectedState, actualState) {
    const inconsistencies = this.detectInconsistencies(expectedState, actualState);
    if (inconsistencies.length > 0) {
      await this.performStateRecovery(inconsistencies);
    }
    return inconsistencies.length === 0;
  }
}
```

### 4. 自动启用管理器 (background/auto-enable-manager.js)
**职责**: 统一自动启用逻辑、域名匹配缓存、性能优化

```javascript
class AutoEnableManager {
  constructor(stateManager) {
    this.stateManager = stateManager;
    this.domainCache = new Map();
    this.cacheExpiry = 5 * 60 * 1000; // 5分钟
    this.debounceDelay = 100;
    this.processingQueue = new Set();
    this.lastCleanup = Date.now();
  }
  
  // 处理标签页更新
  async handleTabUpdate(tabId, url) {
    if (this.processingQueue.has(tabId)) {
      return; // 避免重复处理
    }
    
    this.processingQueue.add(tabId);
    
    try {
      await this.debounce(`tab-${tabId}`, async () => {
        await this.checkAndApplyAutoEnable(tabId, url);
      }, this.debounceDelay);
    } finally {
      this.processingQueue.delete(tabId);
    }
  }
  
  // 智能域名匹配（带缓存）
  async checkDomainMatch(domain, autoEnableList) {
    const cacheKey = `${domain}-${JSON.stringify(autoEnableList)}`;
    const cached = this.domainCache.get(cacheKey);
    
    if (cached && Date.now() - cached.timestamp < this.cacheExpiry) {
      return cached.result;
    }
    
    const result = this.performDomainMatch(domain, autoEnableList);
    this.domainCache.set(cacheKey, {
      result,
      timestamp: Date.now()
    });
    
    return result;
  }
}
```

### 5. 图标管理器 (background/icon-manager.js)
**职责**: 智能图标状态管理、防抖更新、徽章显示

```javascript
class IconManager {
  constructor() {
    this.updateDebounceDelay = 100;
    this.lastUpdateTime = 0;
    this.pendingUpdate = null;
  }
  
  // 智能图标状态计算
  calculateIconState(pluginState) {
    if (!pluginState.pluginEnabled) {
      return 'default';
    }
    
    const enabledFeatures = [];
    if (pluginState.copyEnabled) enabledFeatures.push('copy');
    if (pluginState.pasteEnabled) enabledFeatures.push('paste');
    
    if (enabledFeatures.length === 0) {
      return 'inactive';
    } else if (enabledFeatures.length === 2) {
      return 'active';
    } else {
      return 'partial';
    }
  }
  
  // 防抖图标更新
  async updateIcon(state) {
    if (this.pendingUpdate) {
      clearTimeout(this.pendingUpdate);
    }
    
    this.pendingUpdate = setTimeout(async () => {
      await this.performIconUpdate(state);
      this.lastUpdateTime = Date.now();
      this.pendingUpdate = null;
    }, this.updateDebounceDelay);
  }
}
```

### 6. 监控管理器 (background/monitoring-manager.js)
**职责**: 状态变更日志、性能监控、健康检查

```javascript
class MonitoringManager {
  constructor() {
    this.logs = [];
    this.maxLogEntries = 1000;
    this.performanceMetrics = {
      operationCounts: new Map(),
      responseTimes: [],
      errorCounts: new Map()
    };
    this.healthThresholds = {
      maxResponseTime: 1000,
      maxErrorRate: 0.05,
      maxMemoryUsage: 50 * 1024 * 1024
    };
  }
  
  // 记录状态变更
  async logStateChange(changes, operationId, level = 'info') {
    const logEntry = {
      timestamp: Date.now(),
      level,
      operation: 'state_change',
      operationId,
      changes,
      memoryUsage: this.getMemoryUsage()
    };
    
    this.addLogEntry(logEntry);
    this.updatePerformanceMetrics('state_change');
  }
  
  // 健康检查
  async performHealthCheck() {
    const metrics = this.getPerformanceMetrics();
    const issues = [];
    
    if (metrics.avgResponseTime > this.healthThresholds.maxResponseTime) {
      issues.push('High response time detected');
    }
    
    if (metrics.errorRate > this.healthThresholds.maxErrorRate) {
      issues.push('High error rate detected');
    }
    
    return {
      healthy: issues.length === 0,
      issues,
      metrics,
      timestamp: Date.now()
    };
  }
}
```
